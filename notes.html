<script>

    function generate() {

        const type = document.getElementById('procType').value;

        const extra = document.getElementById('extraNarrative').value;

        const tooth = document.getElementById('toothNum').value || "(#)";

        const surf = document.getElementById('surface').value || "(surf/mat)";

        const anes = document.getElementById('anesCarps').value + " carp " + document.getElementById('anesType').value;

        const manual = document.getElementById('cariesManual').value;

        

        let note = "";

        const consent = "consent: risks/benefits/alternatives discussed; patient consented to treatment and understands all risks.";

        const sig = "\n\nRSiu DMD; Assistant Leng.";


        const cariesText = manual ? `caries noted on chart: ${manual}.` : "caries noted on chart.";


        // Procedure Logic

        if (type === "np_exam") {

            note = `NP EXAM\ntx: np exam, 4bws, pano, 3unit scale, polish/fluoride application.\n\ncc: none. mh: asa 1. ocs: negative for lesions. \nhygiene: fair. eoe/ioe: cl1 occlusion, tmj wnl. ${cariesText}\nxrays: caries noted. bone wnl. justification: screen for pathology. \nperio dx: gingivitis. probing: 1-3mm. \nhyg: piezo, hs, ohi reviewed. polished. fluoride applied. 6mr recommended. ${consent}\nnv: restore.`;

        } else if (type === "composite") {

            note = `COMPOSITE BONDING\ntx: #${tooth} ${surf} composite.\n\ncc: none. mh: asa 1. tx rendered: risks/benefits/post-op sensitivity warning discussed; anes ${anes}; cotton roll isolation; prep w/ hs/ls burs for caries removal; matrix w/ garrison sectional system; bond w/ selective enamel etch, clearfil se primer (20s) and bond, light-cured; restorative w/ filtek supreme incremental application, light-cured; occlusion checked/adjusted; polished. ${consent}\nnv: 6mr.`;

        } else if (type === "pulp_cap") {

            note = `INDIRECT PULP CAP\ntx: #${tooth} ${surf} composite w/ indirect pulp cap.\n\ncc: none. mh: asa 1. tx rendered: risks/benefits/post-op sensitivity warning; warning: advised patient that due to deep decay, rct may be needed if sensitivity persists; anes ${anes}; cotton roll isolation; prep w/ hs/ls burs for caries removal; indirect pulp cap placed w/ (dycal/vitrebond/limelite); bond w/ selective enamel etch, clearfil se primer and bond, light-cured; restorative w/ filtek supreme, light-cured; occlusion checked/adjusted; polished. ${consent}\nnv: monitor.`;

        } else if (type === "crown_prep") {

            note = `CROWN AND CORE BUILDUP\ntx: #${tooth} (${surf}) crown, #${tooth} core buildup.\n\ncc: none. mh: asa 1. tx rendered: risks/benefits/post-op sensitivity warning; anes ${anes}; cotton roll isolation; prep w/ hs/ls burs for caries removal; core buildup placed; crown prep completed; shade selected: ${surf}; digital impression taken w/ trios 3 scanner; temp crown fabricated w/ bis-acrylic provisional material and cemented w/ temp-bond ne; excess cement removed; occlusion adjusted. ${consent}\nnv: crown delivery.`;

        } else if (type === "surg_ext") {

            note = `SURGICAL EXTRACTION\ntx: #${tooth} surgical extraction.\n\ncc: non-restorable. mh: asa 1. tx rendered: risks/benefits/pain/swelling/dry socket/paresthesia discussed; anes ${anes}; mucoperiosteal flap reflected; bone removed w/ surgical hs bur; tooth sectioned and roots delivered; socket curetted; saline irrigation; sutures placed; hemostasis achieved. post-op: poi given; no smoking/straws. ${consent}\nnv: 1wk post-op.`;

        } else if (type === "endo") {

            note = `ENDO TREATMENT\ntx: #${tooth} RCT.\n\ncc: pain/swelling. mh: asa 1. tx rendered: risks/benefits discussed; anes ${anes}; rubber dam isolation; access w/ hs bur; canals located; mechanical instrumentation using Vortex Blue 0.04 taper; irrigation w/ NaOCl and EDTA; ultrasonic activation; canals dried; obturated using single cone technique w/ gutta-percha and bioceramic sealer; orifice sealed; temp restoration. ${consent}\nnv: crown/core.`;

        } else if (type === "recall") {

            note = `RECALL EXAM\ntx: recall exam, 4BWs, 2PAs, 3u scale, polish, fl app.\n\ncc: none. mh: asa 1. xrays: 4BWs and 2PAs taken. eoe/ioe: oral cancer screening negative. perio: gingivitis; 3 units of scaling performed. dental: restorations intact. tx rendered: polish; fluoride applied. ohis: reviewed technique. ${consent}\nnv: 6-month recall.`;

        } else if (type === "prelim_imp") {

            note = `PRELIMINARY IMPRESSION\ntx: prelim imps max/mand for custom trays.\ncc: edentulous. mh: asa 1. dental hx: long term edentulous. ioe: arches edentulous. imps: max/mand alginate in stock trays, good border extension. ${consent}\nnv: final imps.`;

        } else if (type === "cd_del") {

            note = `CD DELIVERY\ntx: delivery of u/l complete dentures. mh: asa 1. ioe: dentures seated; pip paste used to identify pressure spots. occlusion: verified in cr; selective grinding performed. ohi: soak nightly, do not sleep in dentures. ${consent}\nnv: post-op f/u.`;

        } else {

            note = `PROCEDURE: ${type.toUpperCase()}\ntx rendered for tooth #${tooth}. ${consent}`;

        }


        // Additional Narratives

        let narrativeBlock = "";

        if (extra === "bruxism") narrativeBlock = "\n\nnarrative: evidence of bruxism/clenching with significant attrition and wear facets. occlusal guard required to protect against further tooth wear, prevent fracture of teeth/restorations, and manage TMJ discomfort.";

        if (extra === "pulp_narrative") narrativeBlock = `\n\nnarrative: tooth #${tooth} presents with deep caries and a defective existing restoration impacting structural integrity; crown is required to ensure tooth can function and to protect remaining cusps from fracture.`;

        if (extra === "non_restorable") narrativeBlock = `\n\nnarrative: tooth #${tooth} non-restorable/hopeless due to deep subgingival caries, severe lack of tooth structure (no ferrule), and significant periodontal compromise. extraction medically necessary to eliminate infection.`;

        if (extra === "decline_rct") narrativeBlock = `\n\nnarrative: tooth #${tooth} non-restorable due to extensive caries/fracture. patient declined RCT and elects for extraction. tooth is deemed hopeless without endodontic intervention; extraction necessary to eliminate infection.`;


        document.getElementById('output').innerText = note + narrativeBlock + sig;

    }


    function copyText() {

        const text = document.getElementById('output').innerText;

        navigator.clipboard.writeText(text).then(() => {

            alert("Copied to clipboard!");

        });

    }

</script>
