<script>

    function generate() {

        const type = document.getElementById('templateType').value;

        const narr = document.getElementById('narrativeType').value;

        const tooth = document.getElementById('tooth').value || "(tooth #)";

        const detail = document.getElementById('detail').value || "(surface/material)";

        const shade = document.getElementById('shade').value || "(shade)";

        const caries = document.getElementById('cariesManual').value;

        const anesType = document.getElementById('anesType').value;

        const anesCarps = document.getElementById('anesCarps').value;


        const consentLine = "consent: risks and benefits discussed; patient consented to treatment and understands all risks.";

        const sig = "\n\nRSiu DMD; Assistant Leng.";

        

        let note = "";

        let anesText = `anes ${anesCarps} carp ${anesType}`;


        // Narrative Section

        let narrativeBlock = "";

        if(narr === "bruxism") narrativeBlock = "\nnarrative: evidence of bruxism/clenching with significant attrition and wear facets. occlusal guard required to protect against further tooth wear, prevent fracture of teeth/restorations, and manage TMJ discomfort.";

        if(narr === "pulp_narrative") narrativeBlock = `\nnarrative: tooth #${tooth} presents with deep caries and a defective existing restoration impacting structural integrity; crown is required to ensure tooth can function and to protect remaining cusps from fracture.`;

        if(narr === "non_restorable") narrativeBlock = `\nnarrative: tooth #${tooth} non-restorable/hopeless due to deep subgingival caries, severe lack of tooth structure (no ferrule), and significant periodontal compromise. extraction medically necessary to eliminate infection and prevent damage to adjacent teeth.`;

        if(narr === "decline_rct") narrativeBlock = `\nnarrative: tooth #${tooth} non-restorable due to extensive caries/fracture. patient was informed of all clinical options, including endodontic therapy and crown; however, patient declines RCT and elects for extraction. tooth is deemed hopeless without endodontic intervention; extraction necessary to eliminate infection.`;


        let cariesText = caries ? `caries noted on chart: ${caries}.` : "caries noted on chart.";


        switch(type) {

            case "np_exam":

                note = `NP EXAM\ntx: np exam, 4bws, pano, 3unit scale, polish/fluoride application.\n\ncc: none. mh: asa 1. ocs: negative for lesions or lymphadenopathy.\ndental hx: ldv 1 yr ago. hygiene: fair, brush 1x daily, flosses 1x nightly.\neoe/ioe: cl1 occlusion, 2-3mm oj, 50% ob. tmj: wnl. soft tissue wnl. ${cariesText}\nxrays: findings: caries noted. bone wnl. lower anterior radiographic calc. justification: screen for pathology.\nperio dx: gingivitis w/ recession. probing: 1-3mm with localized bop. caries risk: moderate. \nhyg: piezo, hs, ohi reviewed. polished. fluoride applied. 6mr recommended. ${consentLine}\nnv: restore.`;

                break;

            

            case "composite":

                note = `COMPOSITE BONDING\ntx: #${tooth} ${detail} composite.\n\ncc: none. mh: asa 1. tx rendered: risks/benefits/post-op sensitivity warning discussed and consented; ${anesText}; cotton roll isolation; prep w/ hs/ls burs for caries removal; matrix w/ garrison sectional system; bond w/ selective enamel etch, clearfil se primer (20s) and bond, light-cured; restorative w/ filtek supreme incremental application, light-cured; occlusion checked/adjusted; polished. ${consentLine}\nnv: 6mr.`;

                break;


            case "pulp_cap":

                note = `INDIRECT PULP CAP\ntx: #${tooth} ${detail} composite w/ indirect pulp cap.\n\ncc: none. mh: asa 1. tx rendered: risks/benefits/post-op sensitivity warning discussed and consented; warning: advised patient that due to deep decay, rct may be needed if sensitivity persists/worsens; ${anesText}; cotton roll isolation; prep w/ hs/ls burs for caries removal; indirect pulp cap placed w/ (dycal/vitrebond/limelite); matrix w/ garrison sectional system; bond w/ selective enamel etch, clearfil se primer (20s) and bond, light-cured; restorative w/ filtek supreme, light-cured; occlusion checked/adjusted; polished. ${consentLine}\nnv: monitor.`;

                break;


            case "crown_prep":

                note = `CROWN AND CORE BUILDUP\ntx: #${tooth} ${detail} crown, #${tooth} core buildup.\n\ncc: none. mh: asa 1. tx rendered: risks/benefits/post-op sensitivity warning discussed and consented; ${anesType === 'None' ? 'no anesthetic' : anesText}; cotton roll isolation; prep w/ hs/ls burs for caries removal; core buildup placed w/ base, selective enamel etch, clearfil se primer and bond, and filtek supreme composite; crown prep completed; shade selected: ${shade}; digital impression taken w/ trios 3 scanner; temp crown fabricated w/ bis-acrylic provisional material and cemented w/ temp-bond ne; occlusion adjusted. ${consentLine}\nnv: crown delivery.`;

                break;


            case "simple_ext":

                note = `SIMPLE EXTRACTION\ntx: #${tooth} extraction.\n\ncc: reason. mh: asa 1. tx rendered: risks/benefits/post-op sensitivity/paresthesia/fracture of adj teeth discussed and consented; ${anesText}; luxated and delivered w/ elevators and forceps; socket curetted; saline irrigation; hemostasis achieved. post-op: poi given; no smoking/straws/spitting. ${consentLine}\nnv: healing check.`;

                break;


            case "surg_ext":

                note = `SURGICAL EXTRACTION\ntx: #${tooth} surgical extraction.\n\ncc: reason. mh: asa 1. tx rendered: risks/benefits discussed and consented: pain, swelling, infection, dry socket, fracture of adjacent teeth/restorations, and paresthesia; ${anesText}; mucoperiosteal flap reflected; bone removed w/ surgical hs bur under saline irrigation; tooth sectioned and roots delivered; socket curetted/debrided; bone file used; saline irrigation; gut plain sutures placed; hemostasis achieved. post-op: poi given. ${consentLine}\nnv: 1wk f/u.`;

                break;


            case "prelim_imp":

                note = `PRELIMINARY IMPRESSION\ntx: prelim imps max/mand for custom trays. cc: edentulous/missing teeth. mh: asa 1. dental hx: long term edentulous/recent extractions. ioe: u/l arches edentulous, soft tissue wnl. imps: max/mand alginate in stock trays, good border extension, landmarks captured. ${consentLine} nv: final imps.`;

                break;


            case "cd_del":

                note = `CD DELIVERY\ntx: delivery of u/l complete dentures. mh: asa 1. ioe: dentures seated; pip paste used to identify/relieve pressure spots. occlusion: verified in cr; selective grinding performed. stability/retention: good. ohi: brush 2x daily, soak nightly, do not sleep in dentures, 24hr follow-up explained. ${consentLine} nv: post-op f/u.`;

                break;


            case "rct":

                note = `ENDO TREATMENT\ntx: #${tooth} RCT. cc: pain/swelling. mh: asa 1. tx rendered: risks/benefits discussed and consented; ${anesText}; rubber dam isolation; access w/ hs bur; canals located; glide path established; mechanical instrumentation using Vortex Blue; irrigation w/ NaOCl and EDTA; ultrasonic activation; canals dried; obturated using single cone technique w/ gutta-percha and bioceramic sealer; orifice sealed; temp restoration. ${consentLine} nv: crown/core.`;

                break;

            

            // Default handler for others (Spec exam, Recall, etc.)

            default:

                note = `PROCEDURE: ${type.toUpperCase()}\ntx rendered for tooth #${tooth}. ${consentLine}`;

        }


        document.getElementById('output').innerText = note + narrativeBlock + sig;

    }


    function copyNote() {

        const text = document.getElementById('output').innerText;

        navigator.clipboard.writeText(text);

    }

</script>
